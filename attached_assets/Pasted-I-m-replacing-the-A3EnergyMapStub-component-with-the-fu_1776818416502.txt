jsfunction placeItemsOnRing(list, radius) {
  const n = list.length;
  if (n === 0) return;
  const startAngle = -Math.PI / 2 + (Math.PI / n * 0.5);
  list.forEach((it, i) => {
    const baseAngle = startAngle + (i * (2 * Math.PI / n));
    const idHash = hashStr(it.id);
    const radiusJitter = ((idHash % 22) - 11);
    const angleJitter = (((idHash * 7) % 60) - 30) * (Math.PI / 180) * 0.15;
    const r = radius + radiusJitter;
    const a = baseAngle + angleJitter;
    it._x = 360 + r * Math.cos(a);
    it._y = 260 + r * Math.sin(a);
  });
}

function hashStr(s) {
  let h = 0;
  for (let i = 0; i < s.length; i++) h = ((h << 5) - h + s.charCodeAt(i)) | 0;
  return Math.abs(h);
}
