function placeItemsInBand(list, xMin, xMax) {
  const n = list.length;
  if (n === 0) return;
  const bandHeight = 320 - 110;
  const xRange = xMax - xMin;
  const rows = n <= 4 ? 1 : n <= 8 ? 2 : 3;
  const cols = Math.ceil(n / rows);
  const colW = xRange / Math.max(cols, 1);
  const rowH = bandHeight / (rows + 1);
  list.forEach((it, i) => {
    const row = Math.floor(i / cols);
    const col = i % cols;
    const baseX = xMin + colW * (col + 0.5);
    const baseY = 110 + rowH * (row + 1);
    const idHash = hashStr(it.id);
    const jitterX = ((idHash % 18) - 9);
    const jitterY = (((idHash * 11) % 24) - 12);
    it._x = baseX + jitterX;
    it._y = baseY + jitterY;
  });
}
