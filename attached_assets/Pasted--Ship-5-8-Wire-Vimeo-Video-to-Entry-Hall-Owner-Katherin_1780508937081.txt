/**
 * Helper to check whether a video is ready to play
 * (vs. placeholder mode where the card shows but play does nothing).
 */
export function isVideoReady(meta: VideoMeta): boolean {
  if (meta.mediaType === "video") return meta.vimeoId !== null;
  if (meta.mediaType === "audio") return meta.audioUrl !== null;
  return false;
}

/**
 * Helper to build the Vimeo embed URL from meta.
 * Returns null if the video isn't ready (placeholder mode).
 */
export function getVimeoEmbedUrl(meta: VideoMeta): string | null {
  if (meta.vimeoId === null) return null;
  const hashParam = meta.vimeoHash ? `?h=${meta.vimeoHash}` : "";
  return `https://player.vimeo.com/video/${meta.vimeoId}${hashParam}`;
}

## Vimeo iframe spec

The iframe element:

```html
<iframe
  src="https://player.vimeo.com/video/1198147066?h=92f1bb6e8a"
  width="100%"
  height="100%"
  frameBorder="0"
  allow="autoplay; fullscreen; picture-in-picture; clipboard-write"
  allowFullScreen
  title="Welcome to True Ownership"
  style={{
    position: "absolute",
    top: 0,
    left: 0,
    width: "100%",
    height: "100%",
  }}
/>
```

Wrap it in a 16:9 aspect-ratio container so it scales correctly at all viewport widths:

```css
.video-card-player-wrap {
  position: relative;
  width: 100%;
  padding-bottom: 56.25%;  /* 16:9 */
  border-radius: 12px;
  overflow: hidden;
  background: var(--na1-bg);
}
```

Critical: do NOT autoplay. The user must click the play button first. Autoplay creates a hostile experience and violates browser autoplay policies (modern browsers block it without user gesture anyway).

## Mobile pattern (<600px)

On viewports below 600px:
- Card still shows the same paused state
- Clicking play opens the Vimeo player in a **full-screen overlay** (not inline expansion) — the iframe takes the full viewport height/width with a 12px gold close button anchored top-right with 16px safe-area inset
- Tapping the close button or the dark backdrop dismisses the overlay
- Browser back button also dismisses (push a history state on open, pop on close)

This pattern is the same as the existing room overlays in M3/M4. Reuse the existing overlay infrastructure if there is any; otherwise build a thin equivalent.
