const { Pool } = require('pg');

const pool = new Pool({ connectionString: process.env.DATABASE_URL });

async function init() {
  await pool.query(`
    CREATE TABLE IF NOT EXISTS users (
      id SERIAL PRIMARY KEY,
      email TEXT UNIQUE NOT NULL,
      password_hash TEXT,
      role TEXT NOT NULL CHECK (role IN ('owner','client')),
      created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );
  `);
  // For deployments where the column was originally NOT NULL: a freshly invited
  // client has no password until they accept the invite, so allow NULL here.
  await pool.query(`ALTER TABLE users ALTER COLUMN password_hash DROP NOT NULL;`);
  await pool.query(`
    CREATE TABLE IF NOT EXISTS clients (
      id SERIAL PRIMARY KEY,
      name TEXT NOT NULL,
      owner_user_id INTEGER UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
      logo_data_url TEXT,
      created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
      updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );
  `);
  // For deployments that already had the table without the logo column.
  await pool.query(`ALTER TABLE clients ADD COLUMN IF NOT EXISTS logo_data_url TEXT;`);
  await pool.query(`
    CREATE TABLE IF NOT EXISTS strategies (
      client_id INTEGER PRIMARY KEY REFERENCES clients(id) ON DELETE CASCADE,
      data JSONB NOT NULL,
      updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );
  `);
  await pool.query(`
    CREATE TABLE IF NOT EXISTS app_config (
      key TEXT PRIMARY KEY,
      value TEXT NOT NULL
    );
  `);
  // One-time invite tokens: a client receives one of these by email and uses
  // it to set their password on first sign-in (or after the owner re-sends).
  // status: 'pending' until consumed (used_at) or invalidated (expires_at).
  await pool.query(`
    CREATE TABLE IF NOT EXISTS invites (
      id SERIAL PRIMARY KEY,
      user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
      token TEXT UNIQUE NOT NULL,
      expires_at TIMESTAMPTZ NOT NULL,
      used_at TIMESTAMPTZ,
      created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );
  `);
  await pool.query(`CREATE INDEX IF NOT EXISTS invites_user_id_idx ON invites(user_id);`);
}

module.exports = { pool, init };

  function clearErr(){ errEl.classList.remove('show'); errEl.textContent = ''; }

  form.addEventListener('submit', async (e) => {
    e.preventDefault();
    clearErr();
    submit.disabled = true;
    const email = document.getElementById('email').value.trim();
    const password = document.getElementById('password').value;
    try {
      const r = await fetch('/api/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password }),
      });
      const body = await r.json().catch(()=>({}));
      if (!r.ok) {
        showErr(body.error || 'Sign-in failed.');
        submit.disabled = false;
        return;
      }
      window.location.replace(body.role === 'owner' ? '/admin' : '/workspace');
    } catch(err) {
      showErr('Network error. Try again.');
      submit.disabled = false;
    }
  });
})();

function fmtDate(s){
  if (!s) return '—';
  try {
    const d = new Date(s);
    return d.toLocaleDateString(undefined, { month: 'short', day: 'numeric', year: 'numeric' });
  } catch(_) { return '—'; }
}

const path = require('path');
const crypto = require('crypto');
const express = require('express');
const session = require('express-session');
const PgSession = require('connect-pg-simple')(session);
const bcrypt = require('bcryptjs');

const { pool, init } = require('./server/db');
const { INITIAL } = require('./server/initialState');
const { getSessionSecret, ensureSeedOwner, bootstrapOwnerFromEnv } = require('./server/seed');
const { sendInviteEmail, describeEmailConfig } = require('./server/mailer');

// Invites are valid for 7 days. After that the owner has to re-send.
// (The invite/accept-invite flow is kept dormant — the active flow is
// owner-typed passwords plus owner-driven password resets.)
const INVITE_TTL_MS = 7 * 24 * 60 * 60 * 1000;

function buildInviteUrl(req, token) {
  // Trust proxy is on, so req.protocol/host reflect the real public URL
  // (Replit preview, deployment, etc).
  return `${req.protocol}://${req.get('host')}/accept-invite/${token}`;
}

async function issueInviteForUser(userId) {
  // Invalidate any pending invites for this user so the previous link can't
  // also be used after the owner re-sends.
  await pool.query(
    "UPDATE invites SET expires_at = NOW() WHERE user_id=$1 AND used_at IS NULL AND expires_at > NOW()",
    [userId]
  );
  const token = crypto.randomBytes(32).toString('hex');
  const expires = new Date(Date.now() + INVITE_TTL_MS);
  await pool.query(
    'INSERT INTO invites (user_id, token, expires_at) VALUES ($1, $2, $3)',
    [userId, token, expires]
  );
  return { token, expiresAt: expires.toISOString() };
}

const PORT = parseInt(process.env.PORT || '5000', 10);

async function main() {
  await init();
  const seeded = await ensureSeedOwner();
  if (seeded) {
    console.log('\n========================================');
    console.log('  SEED OWNER ACCOUNT CREATED');
    console.log('  Email:    ' + seeded.email);
    console.log('  Password: ' + seeded.password);
    console.log('  (Change this after first sign-in.)');
    console.log('========================================\n');
  }
  try {
    const bootstrapped = await bootstrapOwnerFromEnv();
    if (bootstrapped && bootstrapped.action !== 'noop') {
      console.log('\n========================================');
      console.log('  OWNER BOOTSTRAP (' + bootstrapped.action + ')');
      console.log('  Email: ' + bootstrapped.email);
      console.log('========================================\n');
    }
  } catch (err) {
    console.error('Owner bootstrap failed:', err && err.message ? err.message : err);
  }
  const sessionSecret = await getSessionSecret();

  const app = express();
  app.set('trust proxy', 1);
  app.use(express.json({ limit: '5mb' }));
  app.use(session({
    store: new PgSession({ pool, tableName: 'user_sessions', createTableIfMissing: true }),
    secret: sessionSecret,
    resave: false,
    saveUninitialized: false,
    cookie: {
      httpOnly: true,
      sameSite: 'lax',
      // Secure-by-default in production. Dev/Replit preview is over the
      // workspace proxy where secure cookies aren't set, so we relax there.
      secure: process.env.NODE_ENV === 'production',
      maxAge: 1000 * 60 * 60 * 24 * 30,
    },
  }));

  // ---------- helpers ----------
  // Resolve the user from the DB on every protected request so role changes
  // (promote/demote) and user deletions take effect immediately, even for
  // sessions that are already open. Sets req.user = { id, email, role }.
  async function requireAuth(req, res, next) {
    if (!req.session || !req.session.userId) return res.status(401).json({ error: 'unauthenticated' });
    try {
      const r = await pool.query('SELECT id, email, role FROM users WHERE id=$1', [req.session.userId]);
      if (r.rowCount === 0) {
        req.session.destroy(() => {});
        return res.status(401).json({ error: 'unauthenticated' });
      }
      req.user = r.rows[0];
      next();
    } catch (e) { next(e); }
  }
  function requireOwner(req, res, next) {
    if (!req.user || req.user.role !== 'owner') return res.status(403).json({ error: 'forbidden' });
    next();
  }
  // Wipe a user's active sessions from the connect-pg-simple store so a
  // demoted (or promoted, or deleted) user is forced to sign in again.
  async function destroyUserSessions(userId) {
    try {
      await pool.query(
        "DELETE FROM user_sessions WHERE (sess::jsonb)->>'userId' = $1",
        [String(userId)]
      );
    } catch (e) { console.error('destroyUserSessions failed', e); }
  }

  // ---------- AUTH ----------
  app.post('/api/login', async (req, res) => {
    const { email, password } = req.body || {};
    if (!email || !password) return res.status(400).json({ error: 'Email and password required.' });
    const r = await pool.query('SELECT id, email, password_hash, role FROM users WHERE email=$1', [String(email).toLowerCase().trim()]);
    if (r.rowCount === 0) return res.status(401).json({ error: 'Invalid email or password.' });
    const u = r.rows[0];
    // A freshly invited user has no password_hash yet — they need to accept
    // their invite before they can sign in.
    if (!u.password_hash) {
      return res.status(401).json({ error: 'This account was set up via invite. Use your invite link to set a password first.' });
    }
    const ok = await bcrypt.compare(password, u.password_hash);
    if (!ok) return res.status(401).json({ error: 'Invalid email or password.' });
    req.session.userId = u.id;
    req.session.role = u.role;
    res.json({ id: u.id, email: u.email, role: u.role });
  });

  app.post('/api/logout', (req, res) => {
    req.session.destroy(() => res.json({ ok: true }));
  });

  app.get('/api/me', requireAuth, async (req, res) => {
    let clientId = null;
    if (req.user.role === 'client') {
      const c = await pool.query('SELECT id FROM clients WHERE owner_user_id=$1', [req.user.id]);
      clientId = c.rowCount ? c.rows[0].id : null;
    }
    res.json({ id: req.user.id, email: req.user.email, role: req.user.role, clientId });
  });

  app.post('/api/me/password', requireAuth, async (req, res) => {
    const { currentPassword, newPassword } = req.body || {};
    if (!currentPassword || !newPassword) return res.status(400).json({ error: 'Both passwords required.' });
    if (String(newPassword).length < 8) return res.status(400).json({ error: 'New password must be at least 8 characters.' });
    const r = await pool.query('SELECT password_hash FROM users WHERE id=$1', [req.user.id]);
    const ok = await bcrypt.compare(currentPassword, r.rows[0].password_hash);
    if (!ok) return res.status(401).json({ error: 'Current password is incorrect.' });
    const hash = await bcrypt.hash(newPassword, 10);
    await pool.query('UPDATE users SET password_hash=$1 WHERE id=$2', [hash, req.user.id]);
    res.json({ ok: true });
  });

  // ---------- CLIENTS (owners only for create/list/delete) ----------
  app.get('/api/clients', requireAuth, requireOwner, async (req, res) => {
    // invite_pending = true when the user has never set a password (the only
    // way to sign in is via an active invite link).
    const r = await pool.query(`
      SELECT c.id, c.name, c.logo_data_url, c.created_at, c.updated_at,
             u.email, u.id AS user_id, u.role,
             (u.password_hash IS NULL) AS invite_pending
      FROM clients c JOIN users u ON u.id = c.owner_user_id
      ORDER BY LOWER(c.name)
    `);
    res.json(r.rows);
  });

  // List all owners (so admins can see + demote them).
  app.get('/api/owners', requireAuth, requireOwner, async (req, res) => {
    const r = await pool.query(`
      SELECT u.id, u.email, u.created_at,
             c.id AS client_id, c.name AS client_name
      FROM users u
      LEFT JOIN clients c ON c.owner_user_id = u.id
      WHERE u.role = 'owner'
      ORDER BY LOWER(u.email)
    `);
    res.json(r.rows);
  });

  // Promote/demote a user. Body: { role: 'owner' | 'client' }.
  // Refuses to demote yourself, and refuses to remove the last owner.
  app.post('/api/users/:id/role', requireAuth, requireOwner, async (req, res) => {
    const id = parseInt(req.params.id, 10);
    if (!Number.isFinite(id)) return res.status(400).json({ error: 'Bad id.' });
    const { role } = req.body || {};
    if (role !== 'owner' && role !== 'client') {
      return res.status(400).json({ error: "role must be 'owner' or 'client'." });
    }
    const u = await pool.query('SELECT id, role FROM users WHERE id=$1', [id]);
    if (u.rowCount === 0) return res.status(404).json({ error: 'User not found.' });
    const current = u.rows[0];
    if (current.role === role) return res.json({ ok: true, unchanged: true });
    if (id === req.user.id && role === 'client') {
      return res.status(400).json({ error: "You can't demote yourself." });
    }
    if (current.role === 'owner' && role === 'client') {
      const owners = await pool.query("SELECT COUNT(*)::int AS n FROM users WHERE role='owner'");
      if (owners.rows[0].n <= 1) {
        return res.status(400).json({ error: 'At least one owner must exist.' });
      }
    }
    await pool.query('UPDATE users SET role=$1 WHERE id=$2', [role, id]);
    // Force the affected user out of all open sessions so the new role is
    // enforced immediately on their next request (no waiting for cookie expiry).
    await destroyUserSessions(id);
    res.json({ ok: true });
  });

  app.post('/api/clients', requireAuth, requireOwner, async (req, res) => {
    const { name, email, password } = req.body || {};
    const cleanName = String(name || '').trim();
    const cleanEmail = String(email || '').toLowerCase().trim();
    const cleanPassword = String(password || '');
    if (!cleanName) return res.status(400).json({ error: 'Client name is required.' });
    if (!cleanEmail || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(cleanEmail)) return res.status(400).json({ error: 'A valid sign-in email is required.' });
    if (cleanPassword.length < 8) return res.status(400).json({ error: 'Password must be at least 8 characters.' });

    const exists = await pool.query('SELECT id FROM users WHERE email=$1', [cleanEmail]);
    if (exists.rowCount > 0) return res.status(409).json({ error: 'A user with that email already exists.' });

    // Owner-typed password is stored immediately so the client can sign in
    // straight away with the credentials the owner shares with them.
    const hash = await bcrypt.hash(cleanPassword, 10);
    const c = await pool.connect();
    let client;
    try {
      await c.query('BEGIN');
      const u = await c.query("INSERT INTO users (email, password_hash, role) VALUES ($1, $2, 'client') RETURNING id", [cleanEmail, hash]);
      const userId = u.rows[0].id;
      const cl = await c.query('INSERT INTO clients (name, owner_user_id) VALUES ($1, $2) RETURNING id, name, created_at, updated_at', [cleanName, userId]);
      client = cl.rows[0];
      await c.query('INSERT INTO strategies (client_id, data) VALUES ($1, $2)', [client.id, JSON.stringify(INITIAL)]);
      await c.query('COMMIT');
    } catch (e) {
      await c.query('ROLLBACK');
      console.error(e);
      return res.status(500).json({ error: 'Failed to create client.' });
    } finally {
      c.release();
    }

    res.status(201).json({ ...client, email: cleanEmail });
  });

  app.delete('/api/clients/:id', requireAuth, requireOwner, async (req, res) => {
    const id = parseInt(req.params.id, 10);
    if (!Number.isFinite(id)) return res.status(400).json({ error: 'Bad id.' });
    const r = await pool.query('SELECT owner_user_id FROM clients WHERE id=$1', [id]);
    if (r.rowCount === 0) return res.status(404).json({ error: 'Not found.' });
    const userId = r.rows[0].owner_user_id;
    await pool.query('DELETE FROM users WHERE id=$1', [userId]); // cascades clients + strategies
    await destroyUserSessions(userId);
    res.json({ ok: true });
  });

  // Owner-driven password change for any user (clients or other owners).
  // No current-password challenge — the owner is trusted to reset on behalf
  // of the user. The affected user's open sessions are wiped so they have
  // to sign in again with the new password.
  app.post('/api/users/:id/change-password', requireAuth, requireOwner, async (req, res) => {
    const id = parseInt(req.params.id, 10);
    if (!Number.isFinite(id)) return res.status(400).json({ error: 'Bad id.' });
    const { newPassword } = req.body || {};
    if (!newPassword || String(newPassword).length < 8) {
      return res.status(400).json({ error: 'New password must be at least 8 characters.' });
    }
    const u = await pool.query('SELECT id FROM users WHERE id=$1', [id]);
    if (u.rowCount === 0) return res.status(404).json({ error: 'User not found.' });
    const hash = await bcrypt.hash(String(newPassword), 10);
    await pool.query('UPDATE users SET password_hash=$1 WHERE id=$2', [hash, id]);
    await destroyUserSessions(id);
    res.json({ ok: true });
  });

  // ---------- INVITE ACCEPTANCE (public) ----------
  // The client visits /accept-invite/:token, the page calls these to fetch
  // metadata and submit a new password.
  async function loadInvite(token) {
    if (!token || typeof token !== 'string') return null;
    const r = await pool.query(`
      SELECT i.id, i.user_id, i.expires_at, i.used_at,
             u.email, c.name AS client_name, c.id AS client_id
      FROM invites i
      JOIN users u ON u.id = i.user_id
      LEFT JOIN clients c ON c.owner_user_id = u.id
      WHERE i.token = $1
    `, [token]);
    return r.rowCount === 0 ? null : r.rows[0];
  }

  app.get('/api/invite/:token', async (req, res) => {
    const inv = await loadInvite(req.params.token);
    if (!inv) return res.status(404).json({ error: 'Invite not found.' });
    if (inv.used_at) return res.status(410).json({ error: 'This invite has already been used.' });
    if (new Date(inv.expires_at).getTime() < Date.now()) {
      return res.status(410).json({ error: 'This invite has expired. Ask your owner to send a new one.' });
    }
    res.json({ email: inv.email, clientName: inv.client_name });
  });

  app.post('/api/invite/:token', async (req, res) => {
    const { password } = req.body || {};
    if (!password || String(password).length < 8) {
      return res.status(400).json({ error: 'Password must be at least 8 characters.' });
    }
    const inv = await loadInvite(req.params.token);
    if (!inv) return res.status(404).json({ error: 'Invite not found.' });
    if (inv.used_at) return res.status(410).json({ error: 'This invite has already been used.' });
    if (new Date(inv.expires_at).getTime() < Date.now()) {
      return res.status(410).json({ error: 'This invite has expired. Ask your owner to send a new one.' });
    }
    const hash = await bcrypt.hash(password, 10);
    const c = await pool.connect();
    try {
      await c.query('BEGIN');
      // Atomically claim the invite: only the first concurrent submission
      // can flip used_at from NULL to NOW(). If the row is gone, already
      // used, or expired, RETURNING yields nothing and we bail.
      const claim = await c.query(
        `UPDATE invites
            SET used_at = NOW()
          WHERE token = $1 AND used_at IS NULL AND expires_at > NOW()
          RETURNING id, user_id`,
        [req.params.token]
      );
      if (claim.rowCount === 0) {
        await c.query('ROLLBACK');
        return res.status(410).json({ error: 'This invite is no longer valid. Ask your owner to send a new one.' });
      }
      const claimed = claim.rows[0];
      await c.query('UPDATE users SET password_hash=$1 WHERE id=$2', [hash, claimed.user_id]);
      // Burn any other still-pending invites for this user as well, so a
      // leaked old link can't be re-used after the password is set.
      await c.query(
        'UPDATE invites SET expires_at=NOW() WHERE user_id=$1 AND used_at IS NULL AND id <> $2',
        [claimed.user_id, claimed.id]
      );
      await c.query('COMMIT');
      inv.user_id = claimed.user_id;
    } catch (e) {
      await c.query('ROLLBACK');
      console.error(e);
      return res.status(500).json({ error: 'Could not set password.' });
    } finally {
      c.release();
    }
    // Kick any pre-existing sessions, then sign the new user in.
    await destroyUserSessions(inv.user_id);
    const u = await pool.query('SELECT id, email, role FROM users WHERE id=$1', [inv.user_id]);
    if (u.rowCount === 0) return res.status(500).json({ error: 'Could not load user.' });
    req.session.userId = u.rows[0].id;
    req.session.role = u.rows[0].role;
    res.json({ ok: true, role: u.rows[0].role });
  });

  // ---------- STRATEGY (clients can only read/write their own) ----------
  async function authorizeClientAccess(req, res, next) {
    const id = parseInt(req.params.id, 10);
    if (!Number.isFinite(id)) return res.status(400).json({ error: 'Bad id.' });
    const cr = await pool.query('SELECT id, name, owner_user_id FROM clients WHERE id=$1', [id]);
    if (cr.rowCount === 0) return res.status(404).json({ error: 'Not found.' });
    const client = cr.rows[0];
    // Authoritative role check: req.user is loaded from the DB by requireAuth.
    if (req.user.role === 'owner') {
      req.client = client;
      return next();
    }
    if (req.user.role === 'client' && client.owner_user_id === req.user.id) {
      req.client = client;
      return next();
    }
    return res.status(403).json({ error: 'forbidden' });
  }

  app.get('/api/clients/:id', requireAuth, authorizeClientAccess, async (req, res) => {
    res.json({ id: req.client.id, name: req.client.name });
  });

  app.get('/api/clients/:id/strategy', requireAuth, authorizeClientAccess, async (req, res) => {
    const r = await pool.query('SELECT data, updated_at FROM strategies WHERE client_id=$1', [req.client.id]);
    if (r.rowCount === 0) {
      await pool.query('INSERT INTO strategies (client_id, data) VALUES ($1, $2) ON CONFLICT (client_id) DO NOTHING', [req.client.id, JSON.stringify(INITIAL)]);
      return res.json({ data: INITIAL, updatedAt: new Date().toISOString() });
    }
    res.json({ data: r.rows[0].data, updatedAt: r.rows[0].updated_at });
  });

  app.put('/api/clients/:id/strategy', requireAuth, authorizeClientAccess, async (req, res) => {
    const data = req.body && req.body.data;
    if (!data || typeof data !== 'object') return res.status(400).json({ error: 'Body must be { data: {...} }' });
    const logo = (typeof data.clientLogo === 'string' && data.clientLogo.startsWith('data:'))
      ? data.clientLogo : null;
    await pool.query(
      `INSERT INTO strategies (client_id, data, updated_at)
       VALUES ($1, $2, NOW())
       ON CONFLICT (client_id) DO UPDATE SET data = EXCLUDED.data, updated_at = NOW()`,
      [req.client.id, JSON.stringify(data)]
    );
    // Mirror the logo onto clients.logo_data_url so the admin list can show a thumbnail
    // without loading the full strategy JSON for every row.
    await pool.query('UPDATE clients SET logo_data_url=$1, updated_at = NOW() WHERE id = $2', [logo, req.client.id]);
    res.json({ ok: true, updatedAt: new Date().toISOString() });
  });

  app.post('/api/clients/:id/strategy/reset', requireAuth, authorizeClientAccess, async (req, res) => {
    await pool.query(
      `INSERT INTO strategies (client_id, data, updated_at)
       VALUES ($1, $2, NOW())
       ON CONFLICT (client_id) DO UPDATE SET data = EXCLUDED.data, updated_at = NOW()`,
      [req.client.id, JSON.stringify(INITIAL)]
    );
    // Mirror logo clear so the admin thumbnail also resets.
    await pool.query('UPDATE clients SET logo_data_url=NULL, updated_at=NOW() WHERE id=$1', [req.client.id]);
    res.json({ ok: true });
  });

  // ---------- PAGES ----------
  // Server-side guards so unauthenticated visitors never see admin/workspace HTML.
  // Page guards also resolve role from the DB so promoted/demoted users land
  // on the right page without needing to sign out and back in.
  async function resolveSessionRole(req) {
    if (!req.session || !req.session.userId) return null;
    const r = await pool.query('SELECT role FROM users WHERE id=$1', [req.session.userId]);
    if (r.rowCount === 0) {
      req.session.destroy(() => {});
      return null;
    }
    return r.rows[0].role;
  }
  async function pageRequireAuth(req, res, next) {
    const role = await resolveSessionRole(req);
    if (!role) return res.redirect('/login');
    req.role = role;
    next();
  }
  async function pageRequireOwner(req, res, next) {
    const role = await resolveSessionRole(req);
    if (!role) return res.redirect('/login');
    if (role !== 'owner') return res.redirect('/workspace');
    next();
  }

  app.get('/', async (req, res) => {
    const role = await resolveSessionRole(req);
    if (role === 'owner') return res.redirect('/admin');
    if (role === 'client') return res.redirect('/workspace');
    res.redirect('/login');
  });

  app.get('/login', async (req, res) => {
    const role = await resolveSessionRole(req);
    if (role === 'owner') return res.redirect('/admin');
    if (role === 'client') return res.redirect('/workspace');
    res.sendFile(path.join(__dirname, 'public', 'login.html'));
  });
  app.get('/admin', pageRequireOwner, (req, res) => res.sendFile(path.join(__dirname, 'public', 'admin.html')));
  app.get('/workspace', pageRequireAuth, (req, res) => res.sendFile(path.join(__dirname, 'public', 'workspace.html')));

  // Public invite-acceptance page. The page itself does the token lookup + form
  // submission against /api/invite/:token, so we just serve the static HTML
  // here (no server-side guard — the API enforces the validity checks).
  app.get('/accept-invite/:token', (req, res) => {
    res.sendFile(path.join(__dirname, 'public', 'accept-invite.html'));
  });

  // Tiny favicon stub so logged-out browsers don't spam 404s in the console.
  app.get('/favicon.ico', (_req, res) => res.status(204).end());

  // No express.static() here on purpose: we want every HTML response to go
  // through the role-gated routes above. All CSS/JS is inlined per-page.

  app.use((err, _req, res, _next) => {
    console.error(err);
    res.status(500).json({ error: 'server_error' });
  });

  app.listen(PORT, '0.0.0.0', async () => {
    console.log(`AG45 Strategy server listening on http://0.0.0.0:${PORT}`);
    try {
      const desc = await describeEmailConfig();
      console.log(`Invite email delivery: ${desc}`);
    } catch (e) {
      console.warn('Invite email delivery: could not determine status:', e.message);
    }
  });
}

main().catch(err => {
  console.error('Fatal startup error:', err);
  process.exit(1);
});
